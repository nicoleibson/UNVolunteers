================================================================================
PROJECT LIFELINE CORE — CÓDIGO FUENTE DEL SIMULADOR AMRF v2.0 (INTEGRADO CON RF)
Documento consolidado para NotebookLM
================================================================================

Este documento contiene el código fuente completo de la versión integrada
del simulador AMRF que combina el motor base con los módulos de radio RF.
Incluye 14 paneles (10 base + 4 RF: LoRa Mesh, APRS Tracking, SDR Receiver,
Frequency Allocation) y simulación completa de comunicaciones multi-protocolo.


================================================================================
ARCHIVO 1: index_integrated.html
Estructura HTML5 extendida con 14 paneles. Agrega paneles de LoRa Mesh (915 MHz),
APRS Tracking (144.390 MHz), SDR Receiver (433 MHz), y Frequency Allocation
con canvas para espectro y forma de onda.

// ── Math Helpers ───────────────────────────────────────────────────
function calcDeltaH(pTrapped) {
    return ((R_GAS * T_KELVIN) / (G_GRAVITY * M_MOLAR)) * Math.log(pTrapped / P_START_PA);
}

function calcIPU() {
    const elapsed = (Date.now() - lastTouchTime) / 1000;
    const iTouch  = Math.max(0, 1 - elapsed / 60);
    let   mSig    = 0.1;
    if (activeSOS) mSig += 0.5;
    
    if (watchOnBody) {
        if (bpm > 100 || bpm < 50) mSig += 0.2;
        ipu = Math.min(1.0, Math.max(0.0, 0.25 * iTouch + 0.25 * mSig + 0.50 * 1.0));
    } else {
        ipu = Math.min(1.0, Math.max(0.0, 0.60 + 0.2 * iTouch + 0.2 * mSig));
    }
    return ipu;
}

function getTriageForVitals(bpmV, spo2V, sosFlag, ipuV) {
    if (ipuV < IPU_THRESHOLD) return { code: 3, label: 'BLACK (UNRESPONSIVE)', cls: 'triage-black' };
    if (sosFlag) return { code: 2, label: 'RED (IMMEDIATE)', cls: 'triage-red' };
    
    if (watchOnBody) {
        if (bpmV === 0) return { code: 3, label: 'BLACK (UNRESPONSIVE)', cls: 'triage-black' };
        if (bpmV > 120 || bpmV < 50 || spo2V < 90) return { code: 2, label: 'RED (IMMEDIATE)', cls: 'triage-red' };
        if (bpmV > 100 || bpmV < 60 || spo2V < 95) return { code: 1, label: 'YELLOW (DELAYED)', cls: 'triage-yellow' };
    } else {
        return { code: 1, label: 'YELLOW (UNKNOWN VITALS)', cls: 'triage-yellow' };
    }
    
    return { code: 0, label: 'GREEN (STABLE)', cls: 'triage-green' };
}

function activateLoRaMesh() {
    if (rfState.loraActive) return;
    
    rfState.loraActive = true;
    loraMesh.nodes.forEach(node => {
        node.isActive = true;
        node.lastBeacon = Date.now();
    });
    
    addLog('[LoRa] Mesh network activated — beaconing started', 'log-success');
    addLoRaPacketLog('Mesh network activated — beaconing started');
    
    if (loraStatusEl) {

        loraStatusInd.className = 'status-indicator active';
    }
    
    // Start periodic beaconing
    setInterval(loRaBeaconCycle, 5000);
}

function loRaBeaconCycle() {
    if (!rfState.loraActive) return;
    
    loraMesh.nodes.forEach(node => {
        if (node.isActive) {
            const beacon = {
                type: 'BEACON',
                sourceId: node.id,
                timestamp: Date.now(),
                position: { lat: node.lat, lon: node.lon },
                battery: node.battery,
                signalStrength: -65 - Math.random() * 30,
                hopCount: 0,
            };
            
            loraMesh.packetLog.push(beacon);
            loraMesh.stats.totalPackets++;
            node.lastBeacon = Date.now();
            
            // Add to panel log
            addLoRaPacketLog(`Beacon from ${node.name.slice(0,8)}... (${node.id.slice(0,4)})`);
        }
    });
}

function sendLoRaPacket(sourceId, destinationId, payload) {
    const packet = {
        type: 'DATA',
        sourceId: sourceId,
        destinationId: destinationId || 'BROADCAST',
        timestamp: Date.now(),
        hopCount: 0,
        maxHops: 5,
        payload: payload,
        packetId: `${sourceId}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`,
    };
    
    loraMesh.packetLog.push(packet);
    loraMesh.stats.totalPackets++;
    loraMesh.stats.successfulDeliveries++;
    
    addLoRaPacketLog(`Packet sent: ${sourceId.slice(0,4)} → ${destinationId ? destinationId.slice(0,4) : 'BROADCAST'}`);
    
    // Visualize packet
    particles.push({
        x0: canvas.width * 0.35,
        y0: canvas.height * 0.5,
        x1: canvas.width * 0.8,
        y1: 55,
        progress: 0,
        speed: 0.015,
        color: '#06b6d4',
        type: 'lora',
    });
    
    return packet;
}

function initAPRS() {
    addLog('[APRS] Initializing APRS tracker...', 'log-success');
    addAPRSPacketLog(`Frequency: ${formatFrequency(aprsTracker.frequency)}`);
    
    // Create APRS objects for each victim
    NODES.forEach(node => {
        const lat = parseFloat(node.gps.split(',')[0]);
        const lon = parseFloat(node.gps.split(',')[1]);
        
        aprsTracker.objects.set(node.aprsCallsign, {
            callsign: node.aprsCallsign,
            name: node.label,
            latitude: lat,
            longitude: lon,
            symbol: node.isRelay ? 'R' : '[',
            lastUpdate: 0,
        });
    });
    
    rfState.aprsObjectCount = NODES.length;
    addLog(`[APRS] Tracker ready with ${NODES.length} objects`, 'log-success');
}

function activateAPRS() {
    if (rfState.aprsActive) return;
    
    rfState.aprsActive = true;
    
    addLog('[APRS] APRS tracker activated — beaconing started', 'log-success');
    addAPRSPacketLog('Protocol handler ready — beaconing started');
    
    if (aprsStatusEl) {
        aprsStatusEl.textContent = 'ACTIVE';
        aprsStatusEl.style.color = '#10b981';
    }
    if (aprsStatusInd) {
        aprsStatusInd.className = 'status-indicator simulated';
    }
    
    // Start periodic beacons
    setInterval(aprsBeaconCycle, 30000);
}

function aprsBeaconCycle() {
    if (!rfState.aprsActive) return;
    
    aprsTracker.objects.forEach((obj, callsign) => {
        // Add some movement simulation
        obj.latitude += (Math.random() - 0.5) * 0.0001;
        obj.longitude += (Math.random() - 0.5) * 0.0001;
        
        const packet = {
            type: 'POSITION',
            callsign: callsign,
            timestamp: Date.now(),
            position: { lat: obj.latitude, lon: obj.longitude },
            symbol: obj.symbol,
        };
        
        aprsTracker.stats.packetsSent++;
        aprsTracker.stats.lastBeacon = Date.now();
        obj.lastUpdate = Date.now();
        
        // Add to panel log
        addAPRSPacketLog(`Position beacon: ${callsign} (${obj.latitude.toFixed(4)}, ${obj.longitude.toFixed(4)})`);
    });
}

function sendAPRSMessage(addressee, message) {
    const packet = {
        type: 'MESSAGE',
        callsign: 'N0CALL',
        addressee: addressee,
        message: message,
        timestamp: Date.now(),
    };
    
    aprsTracker.stats.packetsSent++;
    addAPRSPacketLog(`Message to ${addressee}: ${message}`);
    
    return packet;
}

}

function activateSDR() {
    if (rfState.sdrActive) return;
    
    rfState.sdrActive = true;
    sdrState.isReceiving = true;
    
    addLog('[SDR] SDR receiver activated — scanning', 'log-success');
    addSDRPacketLog('Receiver activated — scanning for signals...');
    
    if (sdrStatusEl) {
        sdrStatusEl.textContent = 'RECEIVING';
        sdrStatusEl.style.color = '#10b981';
    }
    if (sdrStatusInd) {
        sdrStatusInd.className = 'status-indicator active';
    }
    
    // Start scanning
    sdrScanLoop();
}

function sdrScanLoop() {
    if (!rfState.sdrActive) return;
    
    // Simulate signal detection
    if (Math.random() < 0.15) {
        const signal = {
            frequency: sdrState.frequency + (Math.random() - 0.5) * 10000,
            strength: -80 + Math.random() * 60,
            bandwidth: 12500,
            type: Math.random() < 0.5 ? 'LoRa' : 'APRS',
            timestamp: Date.now(),
        };
        
        sdrState.signals.push(signal);
        rfState.sdrSignalsDetected++;
        
        addSDRPacketLog(`Signal detected: ${formatFrequency(signal.frequency)} (${signal.type}, ${signal.strength.toFixed(0)} dBm)`);
    }
    
    setTimeout(sdrScanLoop, 200);
}

function rssiPips(rssi) {
    const bars = rssi > -60 ? 4 : rssi > -70 ? 3 : rssi > -80 ? 2 : 1;
    const pips  = Array.from({length: 4}, (_, i) =>
        `<div class="rssi-pip${i < bars ? ' active' : ''}" style="height:${(i+1)*4}px"></div>`
    ).join('');
    return `<div class="rssi-bar">${pips}<span style="margin-left:3px;color:#64748b;font-size:0.58rem">${rssi}</span></div>`;
}

//  CANVAS RENDERING
// ═══════════════════════════════════════════════════════════════════

function draw() {
    if (!ctx || !canvas) {
        requestAnimationFrame(draw);
        return;
    }
    
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Grid
    ctx.strokeStyle = 'rgba(6,182,212,0.03)';
    ctx.lineWidth   = 1;
    for (let x = 0; x < canvas.width; x += 32) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke();
    }
    for (let y = 0; y < canvas.height; y += 32) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke();
    }

    // Rubble block
    const rtop = 90, rheight = canvas.height - 150;
    ctx.fillStyle   = 'rgba(15,23,42,0.5)';
    ctx.strokeStyle = 'rgba(255,255,255,0.05)';
    ctx.lineWidth   = 1;
    ctx.beginPath(); ctx.roundRect(20, rtop, canvas.width - 40, rheight, 4); ctx.fill(); ctx.stroke();

    // Crack lines
    ctx.strokeStyle = 'rgba(255,255,255,0.04)';
    ctx.setLineDash([6,10]);
    [[0.1,0.15,0.85,0.35],[0.3,0.45,0.7,0.65],[0.5,0.2,0.9,0.55]].forEach(([x1,y1,x2,y2]) => {
        ctx.beginPath();
        ctx.moveTo(canvas.width * x1, rtop + rheight * y1);
        ctx.lineTo(canvas.width * x2, rtop + rheight * y2);
        ctx.stroke();
    });
    ctx.setLineDash([]);

    // Scout
    scoutX += 1.2 * scoutDir;
    if (scoutX > canvas.width - 60 || scoutX < 60) scoutDir *= -1;

    const sy = 55;
    ctx.fillStyle = '#06b6d4';
    ctx.shadowColor = '#06b6d4'; ctx.shadowBlur = 14;
    ctx.beginPath(); ctx.arc(scoutX, sy, 11, 0, Math.PI * 2); ctx.fill();
    ctx.shadowBlur = 0;

    // Scanner cone
    if (systemState !== 'DORMANT' && systemState !== 'ORPHAN') {
        const grad = ctx.createLinearGradient(scoutX, sy, scoutX, sy + 200);
        grad.addColorStop(0, 'rgba(6,182,212,0.12)');
        grad.addColorStop(1, 'rgba(6,182,212,0)');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.moveTo(scoutX, sy);
        ctx.lineTo(scoutX - 60, sy + 200);
        ctx.lineTo(scoutX + 60, sy + 200);
        ctx.closePath();
        ctx.fill();
    }

    ctx.fillStyle = '#f1f5f9';
    ctx.font = '9px monospace';
    ctx.fillText('SCOUT / UAV', scoutX - 30, sy - 18);

    // Draw victim nodes
    NODES.forEach((node, idx) => {
        if (node.isRelay && !relayActive) return;
        if (systemState === 'DORMANT' || systemState === 'ORPHAN') return;

        const nx = canvas.width  * node.x;
        const ny = canvas.height * node.y;

        if (idx === 0) {
            const liveDepth  = calcDeltaH(pressureTrapped);
            const depthRatio = Math.min(1.0, Math.max(0, liveDepth / 30));
            node.y = 0.30 + depthRatio * 0.50;
        }

        let fnx = canvas.width  * node.x;
        let fny = canvas.height * node.y;

        if (node.pingUntil && Date.now() < node.pingUntil) {
            fnx += (Math.random() - 0.5) * 8;
            fny += (Math.random() - 0.5) * 8;
        }

        // Relay link
        if (node.isRelay) {
            const prevNode = NODES[idx - 1];
            ctx.strokeStyle = 'rgba(139,92,246,0.35)';
            ctx.setLineDash([5,8]);
            ctx.lineWidth = 1.5;
            ctx.beginPath();
            ctx.moveTo(fnx, fny);
            ctx.lineTo(canvas.width * prevNode.x, canvas.height * prevNode.y);
            ctx.stroke();
            ctx.setLineDash([]);
        }

        let nColor = node.color;
        if (idx === 0) {
            const tr = getTriageForVitals(bpm, spo2, activeSOS, ipu);
            nColor = tr.code === 0 ? '#10b981' : tr.code === 1 ? '#f59e0b' : tr.code === 2 ? '#ef4444' : '#475569';
        }

        const pulse = Math.sin(Date.now() / 300 + idx) * 4;
        ctx.strokeStyle = nColor + '55';
        ctx.lineWidth   = 2;
        ctx.beginPath(); ctx.arc(fnx, fny, 14 + pulse, 0, Math.PI * 2); ctx.stroke();

        ctx.fillStyle   = nColor;
        ctx.shadowColor = nColor; ctx.shadowBlur = 16;
        ctx.beginPath(); ctx.arc(fnx, fny, 7, 0, Math.PI * 2); ctx.fill();
        ctx.shadowBlur  = 0;

        ctx.fillStyle = '#e2e8f0';
        ctx.font      = '8px monospace';
        ctx.fillText(idx === 0 ? `${node.label} (LIVE)` : node.label, fnx + 12, fny + 3);
        ctx.fillStyle = '#64748b';
        ctx.fillText(node.id, fnx + 12, fny + 13);

        // LoRa indicator
        if (rfState.loraActive) {
            ctx.fillStyle = '#06b6d4';
            ctx.font = '7px monospace';
            ctx.fillText(`LoRa:${node.loraId.slice(0,4)}`, fnx + 12, fny + 21);
        }

        ctx.strokeStyle = 'rgba(255,255,255,0.08)';
        ctx.lineWidth   = 1;
        ctx.beginPath(); ctx.moveTo(fnx, rtop); ctx.lineTo(fnx, fny); ctx.stroke();

        // BLE wave emission
        const emitChance = systemState === 'VIGILANCE' ? 0.008
                         : systemState === 'ECO'       ? 0.004
                         :                               0.002;

        if (Math.random() < emitChance) {
            const wColor = node.isRelay ? '#8b5cf6' : (idx === 0 ? '#06b6d4' : '#f59e0b');
            waves.push({ x: fnx, y: fny, r: 5, color: wColor, maxR: 280, speed: 2.5 });
            spawnPacket(fnx, fny, scoutX, sy, nColor);
        }
    });

    // Animate waves
    for (let i = waves.length - 1; i >= 0; i--) {
        const w = waves[i];
        w.r += w.speed;
        const alpha = 1 - w.r / w.maxR;
        ctx.strokeStyle = w.color;
        ctx.lineWidth   = Math.max(0.3, 2 * alpha);
        ctx.shadowColor = w.color;
        ctx.shadowBlur  = 6;
        ctx.globalAlpha = alpha;
        ctx.beginPath(); ctx.arc(w.x, w.y, w.r, 0, Math.PI * 2); ctx.stroke();
        ctx.globalAlpha = 1;
        ctx.shadowBlur  = 0;
        if (w.r >= w.maxR) waves.splice(i, 1);
    }

    // Animate data packets
    for (let i = particles.length - 1; i >= 0; i--) {
        const p   = particles[i];
        p.progress += p.speed;
        const t  = p.progress;
        const px = p.x0 + (p.x1 - p.x0) * t;
        const py = p.y0 + (p.y1 - p.y0) * t - Math.sin(t * Math.PI) * 30;

        ctx.fillStyle   = p.color;
        ctx.shadowColor = p.color; ctx.shadowBlur = 10;
        ctx.beginPath(); ctx.arc(px, py, 3, 0, Math.PI * 2); ctx.fill();
        ctx.shadowBlur  = 0;

        if (t >= 1.0) {
            particles.splice(i, 1);
            packetCount++;
            packetsThisMinute++;

            if (Math.random() < 0.3 && systemState !== 'DORMANT') {
                showCryptoToast();
                const tr = getTriageForVitals(bpm, spo2, activeSOS, ipu);
                const dh = Math.abs(calcDeltaH(pressureTrapped));
                animateECIESCapture(tr, dh);
            }
        }

        if (p.progress > 1.2) particles.splice(i, 1);
    }

    // ── TOP-DOWN MAP ──
    if (tctx && topCanvas.width > 0) {
        tctx.clearRect(0, 0, topCanvas.width, topCanvas.height);
        
        tctx.strokeStyle = 'rgba(255,255,255,0.03)';
        tctx.lineWidth = 1;
        for (let x = 0; x < topCanvas.width; x += 40) {
            tctx.beginPath(); tctx.moveTo(x, 0); tctx.lineTo(x, topCanvas.height); tctx.stroke();
        }
        for (let y = 0; y < topCanvas.height; y += 40) {
            tctx.beginPath(); tctx.moveTo(0, y); tctx.lineTo(topCanvas.width, y); tctx.stroke();
        }
        
        tctx.fillStyle = 'rgba(239, 68, 68, 0.05)';
        tctx.beginPath(); tctx.arc(topCanvas.width/2, topCanvas.height/2, 40 + Math.sin(Date.now()/500)*5, 0, Math.PI*2); tctx.fill();
        tctx.fillStyle = 'rgba(239, 68, 68, 0.3)';
        tctx.font = '8px monospace';
        tctx.fillText("EPICENTER", topCanvas.width/2 - 20, topCanvas.height/2 + 3);
        
        if (systemState !== 'DORMANT' && systemState !== 'ORPHAN') {
            NODES.forEach((node, idx) => {
                if (node.isRelay && !relayActive) return;
                
                const angle = node.x * Math.PI * 2;
                const distance = node.y * topCanvas.height * 0.4;
                
                let tx = topCanvas.width/2 + Math.cos(angle) * distance;
                let ty = topCanvas.height/2 + Math.sin(angle) * distance;
                
                if (node.pingUntil && Date.now() < node.pingUntil) {
                    tx += (Math.random() - 0.5) * 4;
                    ty += (Math.random() - 0.5) * 4;
                }
                
                let nColor = node.color;
                if (idx === 0) {
                    const tr = getTriageForVitals(bpm, spo2, activeSOS, ipu);
                    nColor = tr.code === 0 ? '#10b981' : tr.code === 1 ? '#f59e0b' : tr.code === 2 ? '#ef4444' : '#475569';
                }
                
                tctx.strokeStyle = nColor + '66';
                tctx.lineWidth = 1.5;
                tctx.beginPath(); tctx.arc(tx, ty, 8 + (Math.sin(Date.now()/200 + idx)*2), 0, Math.PI*2); tctx.stroke();
                
                tctx.fillStyle = nColor;
                tctx.beginPath(); tctx.arc(tx, ty, 4, 0, Math.PI*2); tctx.fill();
                
                tctx.fillStyle = '#e2e8f0';
                tctx.font = '9px monospace';
                tctx.fillText(node.id, tx + 8, ty + 3);
            });
            
            tctx.fillStyle = '#06b6d4';
            const sx = topCanvas.width/2 + Math.cos(Date.now()/3000) * (topCanvas.width/3);
            const sy2 = topCanvas.height/2 + Math.sin(Date.now()/2500) * (topCanvas.height/3);
            tctx.beginPath(); tctx.arc(sx, sy2, 5, 0, Math.PI*2); tctx.fill();
            tctx.fillText("UAV", sx + 8, sy2 + 3);
            
            tctx.strokeStyle = 'rgba(6,182,212,0.2)';
            tctx.setLineDash([3, 4]);
            tctx.beginPath(); 
            tctx.moveTo(sx, sy2); 
            tctx.lineTo(topCanvas.width/2, topCanvas.height/2);
            tctx.stroke();
            tctx.setLineDash([]);
        }
    }

    // ── VITALS CHART ──
    drawVitalsSparklines();
    
    // ── RF SPECTRUM ──
    drawRFSpectrum();
    
    // ── RF PANELS ──
    drawLoRaWaveform();
    drawSDRSpectrum();
    drawFreqAllocation();

    requestAnimationFrame(draw);
}

}

// ── Vitals Sparklines ──────────────────────────────────────────────
function drawVitalsSparklines() {
    if (!vctx || !vitalsCanvas || vitalsCanvas.width === 0) return;
    if (vitalsHistory.bpm.length < 2) return;

    const W = vitalsCanvas.width;
    const H = vitalsCanvas.height;
    vctx.clearRect(0, 0, W, H);

    vctx.strokeStyle = 'rgba(255,255,255,0.05)';
    vctx.lineWidth = 1;
    for (let y = H * 0.25; y < H; y += H * 0.25) {
        vctx.beginPath(); vctx.moveTo(0, y); vctx.lineTo(W, y); vctx.stroke();
    }

    function drawLine(data, minVal, maxVal, color) {
        if (data.length < 2) return;
        vctx.strokeStyle = color;
        vctx.lineWidth = 2;
        vctx.shadowColor = color;
        vctx.shadowBlur = 4;
        vctx.beginPath();
        const step = W / (VITALS_HISTORY_MAX - 1);
        data.forEach((val, i) => {
            const x = i * step;
            const normalized = (val - minVal) / (maxVal - minVal);
            const y = H - (normalized * (H - 10)) - 5;
            if (i === 0) vctx.moveTo(x, y);
            else vctx.lineTo(x, y);
        });
        vctx.stroke();
        vctx.shadowBlur = 0;
    }

    drawLine(vitalsHistory.bpm, 30, 180, '#ef4444');
    drawLine(vitalsHistory.spo2, 65, 100, '#3b82f6');
    drawLine(vitalsHistory.battery, 0, 100, '#f59e0b');
}

// ── RF Spectrum Display ────────────────────────────────────────────
function drawRFSpectrum() {
    if (!rfctx || !rfCanvas || rfCanvas.width === 0) return;

    const W = rfCanvas.width;
    const H = rfCanvas.height;
    rfctx.clearRect(0, 0, W, H);

    // Background grid
    rfctx.strokeStyle = 'rgba(255,255,255,0.03)';
    rfctx.lineWidth = 1;
    for (let x = 0; x < W; x += 20) {
        rfctx.beginPath(); rfctx.moveTo(x, 0); rfctx.lineTo(x, H); rfctx.stroke();
    }

    // Generate spectrum data
    const spectrum = [];
    for (let i = 0; i < W; i++) {
        let amplitude = Math.random() * 10; // Noise floor
        
        // Add signal peaks
        if (rfState.loraActive && Math.abs(i - W * 0.3) < 10) {
            amplitude += 40 + Math.sin(Date.now() / 200) * 10;
        }
        if (rfState.aprsActive && Math.abs(i - W * 0.6) < 8) {
            amplitude += 30 + Math.sin(Date.now() / 300) * 5;
        }
        
        spectrum.push(amplitude);
    }

    // Draw spectrum
    rfctx.strokeStyle = '#06b6d4';
    rfctx.lineWidth = 1.5;
    rfctx.beginPath();
    spectrum.forEach((amp, i) => {
        const y = H - (amp / 80 * H);
        if (i === 0) rfctx.moveTo(i, y);
        else rfctx.lineTo(i, y);
    });
    rfctx.stroke();

    // Labels
    rfctx.fillStyle = '#64748b';
    rfctx.font = '8px monospace';
    rfctx.fillText('RF SPECTRUM', 5, 12);
    rfctx.fillText(formatFrequency(rfState.currentFrequency), 5, H - 5);
    
    if (rfState.loraActive) {
        rfctx.fillStyle = '#06b6d4';
        rfctx.fillText('● LoRa', W - 60, 12);
    }
    if (rfState.aprsActive) {
        rfctx.fillStyle = '#10b981';
        rfctx.fillText('● APRS', W - 60, 24);
    }
}

// ── SDR Spectrum Drawing ───────────────────────────────────────────

    const W = sdrSpectrumCanvas.width;
    const H = sdrSpectrumCanvas.height;
    sdrSpectrumCtx.clearRect(0, 0, W, H);

    // Background grid
    sdrSpectrumCtx.strokeStyle = 'rgba(255,255,255,0.03)';
    sdrSpectrumCtx.lineWidth = 1;
    for (let x = 0; x < W; x += 20) {
        sdrSpectrumCtx.beginPath(); sdrSpectrumCtx.moveTo(x, 0); sdrSpectrumCtx.lineTo(x, H); sdrSpectrumCtx.stroke();

    if (!rfState.sdrActive) {

        sdrSpectrumCtx.font = '10px monospace';
        sdrSpectrumCtx.fillText('SDR INACTIVE', W/2 - 35, H/2 + 3);
        return;
    }

    // Generate spectrum with signals
    const spectrum = [];
    for (let i = 0; i < W; i++) {
        let amplitude = Math.random() * 8; // Noise floor
        
        // Add detected signal peaks
        sdrState.signals.forEach((sig, idx) => {
            const sigX = W * 0.3 + idx * 30;
            if (Math.abs(i - sigX) < 15) {
                amplitude += Math.abs(sig.strength) * 0.5 + Math.sin(Date.now() / 200 + idx) * 5;
            }
        });
        
        spectrum.push(amplitude);
    }

    // Draw waterfall-style spectrum
    sdrSpectrumCtx.strokeStyle = '#a855f7';
    sdrSpectrumCtx.lineWidth = 1.5;
    sdrSpectrumCtx.beginPath();
    spectrum.forEach((amp, i) => {
        const y = H - (amp / 100 * H);
        if (i === 0) sdrSpectrumCtx.moveTo(i, y);
        else sdrSpectrumCtx.lineTo(i, y);
    });
    sdrSpectrumCtx.stroke();

    // Labels
    sdrSpectrumCtx.fillStyle = '#a855f7';
    sdrSpectrumCtx.font = '8px monospace';
    sdrSpectrumCtx.fillText(`SDR ${formatFrequency(sdrState.frequency)}`, 5, 12);

        } else {
            clearInterval(timerInterval);
            document.getElementById('vigilance-timer').textContent = 'EXPIRED';
            addLog('[System] Vigilance window expired.', 'log-alert');
        }
    }, 1000);

    // Ephemeral ID rotation
    setInterval(() => {
        if (systemState !== 'DORMANT') {
            ephemeralId = randomHex(4);
            NODES.forEach(n => { n.id = randomHex(4); });
            addLog(`[StrongBox] MAC + Ephemeral ID rotated → ${ephemeralId}`, 'log-line');
        }
    }, ID_ROTATE_MS);

});

setInterval(() => {

    }

    envState.temp = Math.max(12, 22 - depth * 0.4 + (Math.random() - 0.5) * 0.3);
    envState.light = Math.max(0, Math.min(50, 50 - depth * 3 + (Math.random() - 0.5) * 2));
    envState.mic = activeSOS ? 70 + Math.random() * 15 : (Math.random() < 0.1 ? 20 + Math.random() * 10 : 5 + Math.random() * 8);
    envState.tilt = Math.min(90, depth * 2.5 + (Math.random() - 0.5) * 3);
    
    // Refresh telemetry panels with the latest samples
    updateVitalsPanel();
    updateEnvironPanel();
}, 1000);

// Vitals panel update
function updateVitalsPanel() {
    const isActive = systemState !== 'DORMANT' && systemState !== 'ORPHAN';
    const statusEl = document.getElementById('vitals-status');
    if (statusEl) {
        statusEl.textContent = isActive ? 'RECORDING' : 'OFFLINE';
        statusEl.style.color = isActive ? 'var(--danger)' : 'var(--muted)';
        statusEl.style.borderColor = isActive ? 'var(--danger)' : 'var(--muted)';
    }
    if (!isActive) return;

    const bpmArr = vitalsHistory.bpm;
    const bpmTrendEl = document.getElementById('vitals-bpm-trend');
    if (bpmTrendEl && bpmArr.length >= 5) {
        const recent = bpmArr.slice(-5);
        const older  = bpmArr.slice(-10, -5);
        if (older.length > 0) {
            const avgRecent = recent.reduce((a,b) => a+b, 0) / recent.length;
            const avgOlder  = older.reduce((a,b) => a+b, 0) / older.length;
            const diff = avgRecent - avgOlder;
            if (diff > 3) { bpmTrendEl.textContent = `${Math.round(avgRecent)} ↑ RISING`; bpmTrendEl.style.color = '#ef4444'; }
            else if (diff < -3) { bpmTrendEl.textContent = `${Math.round(avgRecent)} ↓ FALLING`; bpmTrendEl.style.color = '#f59e0b'; }
            else { bpmTrendEl.textContent = `${Math.round(avgRecent)} → STABLE`; bpmTrendEl.style.color = '#10b981'; }
        } else {
            bpmTrendEl.textContent = `${Math.round(recent[recent.length-1])} ...`;
        }
    }

    const spo2Arr = vitalsHistory.spo2;
    const spo2TrendEl = document.getElementById('vitals-spo2-trend');
    if (spo2TrendEl && spo2Arr.length >= 5) {
        const recent = spo2Arr.slice(-5);
        const avgRecent = recent.reduce((a,b) => a+b, 0) / recent.length;
        if (avgRecent < 90) { spo2TrendEl.textContent = `${avgRecent.toFixed(1)}% ⚠ CRITICAL`; spo2TrendEl.style.color = '#ef4444'; }
        else if (avgRecent < 95) { spo2TrendEl.textContent = `${avgRecent.toFixed(1)}% ↓ LOW`; spo2TrendEl.style.color = '#f59e0b'; }
        else { spo2TrendEl.textContent = `${avgRecent.toFixed(1)}% → NORMAL`; spo2TrendEl.style.color = '#10b981'; }
    }

    const timeEl = document.getElementById('vitals-time-trapped');
    if (timeEl && trappedSince > 0) {
        const elapsed = Math.floor((Date.now() - trappedSince) / 1000);
        const h = Math.floor(elapsed / 3600).toString().padStart(2, '0');
        const m = Math.floor((elapsed % 3600) / 60).toString().padStart(2, '0');
        const s = (elapsed % 60).toString().padStart(2, '0');
        timeEl.textContent = `${h}:${m}:${s}`;
    }

    const etaEl = document.getElementById('vitals-battery-eta');
    if (etaEl) {
        if (autoDrain && battery > 1) {
            const hoursLeft = (battery / (1/3)) / 3600;
            etaEl.textContent = `~${(hoursLeft * 3600 / 60).toFixed(0)} min`;
            etaEl.style.color = battery < 20 ? '#ef4444' : battery < 40 ? '#f59e0b' : 'var(--primary)';
        } else {
            etaEl.textContent = autoDrain ? 'DEPLETED' : 'STATIC';
            etaEl.style.color = 'var(--muted)';
        }
    }
}

// Environment panel update
function updateEnvironPanel() {
    const isActive = systemState !== 'DORMANT' && systemState !== 'ORPHAN';
    if (!isActive) {
        ['env-temp','env-light','env-mic','env-tilt'].forEach(id => {
            const el = document.getElementById(id);
            if (el) el.textContent = '—';
        });
        const ctEl = document.getElementById('collapse-type');
        if (ctEl) { ctEl.textContent = 'UNKNOWN'; ctEl.className = 'collapse-type-badge'; }
        const cdEl = document.getElementById('collapse-desc');
        if (cdEl) cdEl.textContent = 'Awaiting accelerometer data...';
        const apFill = document.getElementById('air-pocket-fill');
        if (apFill) apFill.style.width = '0%';
        const apVal = document.getElementById('air-pocket-val');
        if (apVal) apVal.textContent = '—';
        return;
    }

    const tempEl = document.getElementById('env-temp');
    if (tempEl) {
        tempEl.textContent = `${envState.temp.toFixed(1)}°C`;
        tempEl.style.color = envState.temp < 16 ? '#3b82f6' : envState.temp > 35 ? '#ef4444' : '#10b981';
    }
    const lightEl = document.getElementById('env-light');
    if (lightEl) {
        lightEl.textContent = `${envState.light.toFixed(0)} lux`;
        lightEl.style.color = envState.light < 5 ? '#ef4444' : envState.light < 20 ? '#f59e0b' : '#10b981';
    }
    const micEl = document.getElementById('env-mic');
    if (micEl) {
        const micLabel = envState.mic > 50 ? 'VOICE/TAP' : envState.mic > 15 ? 'AMBIENT' : 'SILENT';
        micEl.textContent = `${envState.mic.toFixed(0)} dB (${micLabel})`;
        micEl.style.color = envState.mic > 50 ? '#10b981' : envState.mic > 15 ? '#f59e0b' : '#ef4444';
    }
    const tiltEl = document.getElementById('env-tilt');
    if (tiltEl) {
        tiltEl.textContent = `${envState.tilt.toFixed(1)}°`;
        tiltEl.style.color = envState.tilt > 45 ? '#ef4444' : envState.tilt > 20 ? '#f59e0b' : '#10b981';
    }

    const depth = Math.abs(calcDeltaH(pressureTrapped));
    const ctEl = document.getElementById('collapse-type');
    const cdEl = document.getElementById('collapse-desc');
    if (ctEl && cdEl) {
        if (envState.tilt > 50) {
            ctEl.textContent = 'PANCAKE';
            ctEl.className = 'collapse-type-badge type-pancake';
            cdEl.textContent = 'Floors collapsed flat.';
        } else if (envState.tilt > 20) {
            ctEl.textContent = 'LEAN-TO';
            ctEl.className = 'collapse-type-badge type-leanto';
            cdEl.textContent = 'Angled collapse. Possible void spaces.';
        } else {
            ctEl.textContent = 'V-SHAPE';
            ctEl.className = 'collapse-type-badge type-vshape';
            cdEl.textContent = 'Best survival probability.';
        }
    }

    const airPocketScore = Math.min(100, Math.max(0, envState.light * 1.5 + (100 - envState.tilt) * 0.3 + (envState.mic > 15 ? 15 : 0)));
    const apFill = document.getElementById('air-pocket-fill');
    const apVal = document.getElementById('air-pocket-val');
    if (apFill) apFill.style.width = `${airPocketScore}%`;
    if (apVal) {
        const label = airPocketScore > 60 ? 'GOOD' : airPocketScore > 30 ? 'MODERATE' : 'CRITICAL';
        apVal.textContent = `${airPocketScore.toFixed(0)}% (${label})`;
        apVal.style.color = airPocketScore > 60 ? '#10b981' : airPocketScore > 30 ? '#f59e0b' : '#ef4444';
    }
}
